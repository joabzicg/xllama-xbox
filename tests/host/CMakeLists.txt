cmake_minimum_required(VERSION 3.16)
project(xllama_host_contracts LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
foreach(name IN ITEMS sse vision model_paths)
    add_executable(verify_${name} "${ROOT}/tools/verify_${name}.cpp")
    target_include_directories(verify_${name} PRIVATE "${ROOT}/include")
    target_link_libraries(verify_${name} PRIVATE Threads::Threads)
    add_test(NAME ${name} COMMAND verify_${name})
endforeach()
target_sources(verify_model_paths PRIVATE "${ROOT}/src/bridge/path_utils.cpp")

# Uses only the pinned headers and a fake backend; never loads a model.
if(EXISTS "${ROOT}/llama.cpp/tools/mtmd/mtmd-helper.h")
    add_executable(verify_vision_runtime "${ROOT}/tools/verify_vision_runtime.cpp"
        "${ROOT}/uwp/vision_mtmd.cpp" "${ROOT}/src/bridge/chat_prompt.cpp")
    target_include_directories(verify_vision_runtime PRIVATE "${ROOT}/include" "${ROOT}/uwp"
        "${ROOT}/src/bridge" "${ROOT}/llama.cpp/include" "${ROOT}/llama.cpp/ggml/include"
        "${ROOT}/llama.cpp/tools/mtmd")
    target_compile_definitions(verify_vision_runtime PRIVATE XLLAMA_HAS_MTMD=1)
    add_test(NAME vision_runtime_contract COMMAND verify_vision_runtime)
    add_library(session_compile_contract OBJECT "${ROOT}/src/bridge/session.cpp")
    target_include_directories(session_compile_contract PRIVATE "${ROOT}/include"
        "${ROOT}/llama.cpp/include" "${ROOT}/llama.cpp/ggml/include" "${ROOT}/llama.cpp/common")
    target_compile_definitions(session_compile_contract PRIVATE XLLAMA_USE_LLAMA=1)
endif()
get_property(checks DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(check IN LISTS checks)
    if(MSVC)
        target_compile_options(${check} PRIVATE /UNDEBUG /utf-8)
    else()
        target_compile_options(${check} PRIVATE -UNDEBUG)
    endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME lan_client_parser COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tests/test_lan_validate.py")
endif()
